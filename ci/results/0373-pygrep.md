  103 def lua_str(s):
  104     """Lua 5.1 string literal; raw UTF-8 bytes kept, control chars as decimal escapes."""
  105     out = ['"']
  106     for ch in s:
  107         o = ord(ch)

  110         elif ch == '"':
  111             out.append('\\"')
  112         elif ch == "\n":
